import Foundation

let root = URL(fileURLWithPath: CommandLine.arguments[1], isDirectory: true)
let files = FileManager.default.enumerator(
    at: root, includingPropertiesForKeys: nil, options: [.skipsHiddenFiles]
)!
var checked = 0
var needsPassword = 0
for case let url as URL in files where url.pathExtension.lowercased() == "3105" {
    do {
        let data = try Data(contentsOf: url)
        let summary = try PatchPackageCodec.inspect(data)
        do {
            let decoded = try PatchPackageCodec.decode(data, password: "@osinxshop")
            precondition(decoded.project.id == summary.packageID)
            checked += 1
        } catch PatchPackageError.invalidPasswordOrCorruptedPackage {
            // Keep packages with another password. The app prompts for it on opening.
            needsPassword += 1
            print("PASSWORD REQUIRED: \(url.lastPathComponent)")
        }
    } catch {
        fputs("FAIL: \(url.lastPathComponent) · \(error)\n", stderr)
        exit(1)
    }
}
guard checked > 0 else {
    fputs("FAIL: no readable bundled packages\n", stderr)
    exit(1)
}
print("PASS: \(checked) packages decoded, \(needsPassword) require their own password")
