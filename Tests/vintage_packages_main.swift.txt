import Foundation

let root = URL(fileURLWithPath: CommandLine.arguments[1], isDirectory: true)
let files = FileManager.default.enumerator(
    at: root, includingPropertiesForKeys: nil, options: [.skipsHiddenFiles]
)!
var checked = 0
var needsPassword = 0
for case let url as URL in files where ["3105", "3106"].contains(url.pathExtension.lowercased()) {
    do {
        let data = try Data(contentsOf: url)
        let summary = try PatchPackageCodec.inspect(data)
        do {
            let decoded = try PatchPackageCodec.decode(
                data,
                password: PatchPackageCodec.bundledPassword(for: summary.packageID)
            )
            precondition(decoded.project.id == summary.packageID)
            if url.deletingLastPathComponent().lastPathComponent == "FPSBoost" {
                let requiredNames = [
                    "BODY VINTAGELT.3105": "BODY VINTAGE",
                    "DRAG VINTAGEE.3105": "DRAG VINTAGEE",
                    "NECK VINTAGE.3105": "NECK VINTAGE"
                ]
                if let expected = requiredNames[url.lastPathComponent] {
                    precondition(
                        PatchPackageCodec.bundledDisplayName(for: summary.packageID) == expected,
                        "Incorrect bundled Boost display name"
                    )
                }
            }
            if url.lastPathComponent == "DRAG VINTAGEE.3105" {
                let originalURL = url.deletingLastPathComponent().appendingPathComponent("NECK VINTAGE.3105")
                let originalData = try Data(contentsOf: originalURL)
                let originalSummary = try PatchPackageCodec.inspect(originalData)
                let original = try PatchPackageCodec.decode(
                    originalData,
                    password: PatchPackageCodec.bundledPassword(for: originalSummary.packageID)
                )
                precondition(decoded.project.id != original.project.id, "The copy must have a separate identity")
                precondition(decoded.project.name == "DRAG VINTAGEE")
                precondition(decoded.project.rules == original.project.rules, "Copied replacement bytes must be preserved")
                precondition(decoded.project.directories == original.project.directories)
                precondition(decoded.project.bundleIdentifiers == original.project.bundleIdentifiers)
            }
            checked += 1
        } catch PatchPackageError.invalidPasswordOrCorruptedPackage {
            // The bundled Boost catalog must open with its configured passwords.
            if url.deletingLastPathComponent().lastPathComponent == "FPSBoost" {
                throw PatchPackageError.invalidPasswordOrCorruptedPackage
            }
            // Keep packages with another password. The app prompts for it on opening.
            needsPassword += 1
            print("PASSWORD REQUIRED: \(url.lastPathComponent)")
        }
    } catch {
        fputs("FAIL: \(url.lastPathComponent) · \(error)\n", stderr)
        exit(1)
    }
}
guard checked + needsPassword > 0 else {
    fputs("FAIL: no bundled patch packages were found\n", stderr)
    exit(1)
}
print("PASS: \(checked) packages decoded, \(needsPassword) require their own password")
